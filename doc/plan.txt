 Решение

 Исправление 1: SqlVariantQueryGenerator - правильный парсинг таблиц

 Задача: Корректно обрабатывать имена таблиц в форматах:
 - "table" → schema="dbo", table="table"
 - "schema.table" → schema="schema", table="table"
 - "database.schema.table" → schema="schema", table="table" (игнорируем database, т.к. подключение уже к нужной БД)

 Реализация (строки 51-60):

 public String generateSelectQuery(String tableName) {
     // Parse schema and table name (supports database.schema.table format)
     String schema = "dbo"; // default schema for SQL Server
     String table = tableName;

     if (tableName.contains(".")) {
         String[] parts = tableName.split("\\.");
         if (parts.length == 2) {
             // Format: schema.table
             schema = parts[0];
             table = parts[1];
         } else if (parts.length >= 3) {
             // Format: database.schema.table (or more parts)
             // Take second-to-last as schema, last as table
             schema = parts[parts.length - 2];
             table = parts[parts.length - 1];
         }
     }

     log.debug("Generating SELECT query for table: schema={}, table={} (original={})",
               schema, table, tableName);

 Примеры работы:
 - "type_conversion_src" → schema="dbo", table="type_conversion_src"
 - "dbo.type_conversion_src" → schema="dbo", table="type_conversion_src"
 - "SUPPORT.dbo.type_conversion_src" → schema="dbo", table="type_conversion_src" ✅

 Исправление 2: AvroToRecordTransformer - извлечение precision/scale

 Задача: Для Avro decimal logical type извлекать precision и scale из схемы

 Реализация (изменить метод buildMetadataFromAvroSchema, строки 60-79):

 private Map<String, ColumnMetadata> buildMetadataFromAvroSchema(Schema avroSchema) {
     Map<String, ColumnMetadata> metadata = new LinkedHashMap<>();

     for (Schema.Field field : avroSchema.getFields()) {
         Schema fieldSchema = unwrapNullable(field.schema());
         String sqlTypeName = avroTypeToSqlTypeName(fieldSchema);
         int jdbcType = sqlTypeNameToJdbcType(sqlTypeName);

         // Extract precision and scale for decimal types
         int precision = 0;
         int scale = 0;
         LogicalType logicalType = fieldSchema.getLogicalType();
         if (logicalType instanceof LogicalTypes.Decimal decimalType) {
             precision = decimalType.getPrecision();
             scale = decimalType.getScale();
         }

         metadata.put(field.name(), new ColumnMetadata(
             field.name(),
             jdbcType,
             sqlTypeName,
             precision,  // Now extracts from Avro decimal
             scale,      // Now extracts from Avro decimal
             field.schema().isNullable()
         ));
     }

     return metadata;
 }

 Обоснование:
 - Avro decimal хранит precision/scale в LogicalTypes.Decimal
 - Без этих значений SQL Server не может создать корректный DECIMAL тип для bulk copy
 - Для остальных типов precision/scale = 0 корректен (например, для INT, VARCHAR без указания длины)
