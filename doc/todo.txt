      result.show()

      # Cell 4: Запись как Parquet
      result.write \
          .mode("overwrite") \
          .parquet("s3a://datalake/analytics/daily_orders/")

      # Cell 5: Проверка Hive Metastore интеграции
      spark.sql("SHOW DATABASES").show()
      spark.sql("SHOW TABLES").show()

      # Cell 6: Создание Delta таблицы
      df.write \
          .format("delta") \
          .mode("overwrite") \
          .save("s3a://datalake/delta/orders/")

      # Cell 7: Чтение Delta таблицы
      delta_df = spark.read.format("delta").load("s3a://datalake/delta/orders/")
      print(f"Delta table rows: {delta_df.count()}")

      # Cell 8: Delta Time Travel (версионирование)
      from delta.tables import DeltaTable
      delta_table = DeltaTable.forPath(spark, "s3a://datalake/delta/orders/")
      print(delta_table.history().show())
